using XFramework.Application.Contracts.Authorization;
using XFramework.Application.Contracts.Services;

namespace XFramework.Application.Services;

public abstract class CrudAppService<TEntity, TDto, TKey, TCreateDto, TUpdateDto>
    : ApplicationService
    where TEntity : class
{
    protected IRepository<TEntity, TKey> Repository { get; }

    protected CrudAppService(
        ICurrentUser currentUser,
        IRepository<TEntity, TKey> repository)
        : base(currentUser)
    {
        Repository = repository;
    }

    // =========================================================
    // GET
    // =========================================================

    public virtual async Task<TDto> GetAsync(
        TKey id,
        CancellationToken cancellationToken = default)
    {
        var entity = await Repository.GetAsync(
            id,
            cancellationToken);

        if (entity is null)
        {
            throw new KeyNotFoundException(
                $"{typeof(TEntity).Name} with id '{id}' was not found.");
        }

        return await MapToDtoAsync(
            entity,
            cancellationToken);
    }

    // =========================================================
    // GET LIST
    // =========================================================

    public virtual async Task<PagedResult<TDto>> GetListAsync(
        PagedRequest request,
        CancellationToken cancellationToken = default)
    {
        var query = Repository.GetQueryable();

        query = await ApplyQueryAsync(
            query,
            request,
            cancellationToken);

        var totalCount = await Repository.CountAsync(
            query,
            cancellationToken);

        query = ApplyPaging(
            query,
            request);

        var entities = await Repository.ToListAsync(
            query,
            cancellationToken);

        var items = new List<TDto>(
            entities.Count);

        foreach (var entity in entities)
        {
            items.Add(
                await MapToDtoAsync(
                    entity,
                    cancellationToken));
        }

        return new PagedResult<TDto>(
            items,
            totalCount,
            request.SkipCount,
            request.MaxResultCount);
    }

    // =========================================================
    // CREATE
    // =========================================================

    public virtual async Task<TDto> CreateAsync(
        TCreateDto input,
        CancellationToken cancellationToken = default)
    {
        await ValidateCreateAsync(
            input,
            cancellationToken);

        await BeforeCreateAsync(
            input,
            cancellationToken);

        var entity = await MapToEntityAsync(
            input,
            cancellationToken);

        await Repository.AddAsync(
            entity,
            cancellationToken);

        await AfterCreateAsync(
            entity,
            cancellationToken);

        return await MapToDtoAsync(
            entity,
            cancellationToken);
    }

    // =========================================================
    // UPDATE
    // =========================================================

    public virtual async Task<TDto> UpdateAsync(
        TKey id,
        TUpdateDto input,
        CancellationToken cancellationToken = default)
    {
        var entity = await Repository.GetAsync(
            id,
            cancellationToken);

        if (entity is null)
        {
            throw new KeyNotFoundException(
                $"{typeof(TEntity).Name} with id '{id}' was not found.");
        }

        await ValidateUpdateAsync(
            entity,
            input,
            cancellationToken);

        await BeforeUpdateAsync(
            entity,
            input,
            cancellationToken);

        await MapToEntityAsync(
            input,
            entity,
            cancellationToken);

        await Repository.UpdateAsync(
            entity,
            cancellationToken);

        await AfterUpdateAsync(
            entity,
            cancellationToken);

        return await MapToDtoAsync(
            entity,
            cancellationToken);
    }

    // =========================================================
    // DELETE
    // =========================================================

    public virtual async Task DeleteAsync(
        TKey id,
        CancellationToken cancellationToken = default)
    {
        var entity = await Repository.GetAsync(
            id,
            cancellationToken);

        if (entity is null)
        {
            throw new KeyNotFoundException(
                $"{typeof(TEntity).Name} with id '{id}' was not found.");
        }

        await BeforeDeleteAsync(
            entity,
            cancellationToken);

        await Repository.DeleteAsync(
            entity,
            cancellationToken);

        await AfterDeleteAsync(
            entity,
            cancellationToken);
    }

    // =========================================================
    // QUERY HOOK
    // =========================================================

    protected virtual Task<IQueryable<TEntity>> ApplyQueryAsync(
        IQueryable<TEntity> query,
        PagedRequest request,
        CancellationToken cancellationToken)
    {
        return Task.FromResult(query);
    }

    protected virtual IQueryable<TEntity> ApplyPaging(
        IQueryable<TEntity> query,
        PagedRequest request)
    {
        return query
            .Skip(request.SkipCount)
            .Take(request.MaxResultCount);
    }

    // =========================================================
    // VALIDATION HOOKS
    // =========================================================

    protected virtual Task ValidateCreateAsync(
        TCreateDto input,
        CancellationToken cancellationToken)
    {
        return Task.CompletedTask;
    }

    protected virtual Task ValidateUpdateAsync(
        TEntity entity,
        TUpdateDto input,
        CancellationToken cancellationToken)
    {
        return Task.CompletedTask;
    }

    // =========================================================
    // CREATE HOOKS
    // =========================================================

    protected virtual Task BeforeCreateAsync(
        TCreateDto input,
        CancellationToken cancellationToken)
    {
        return Task.CompletedTask;
    }

    protected virtual Task AfterCreateAsync(
        TEntity entity,
        CancellationToken cancellationToken)
    {
        return Task.CompletedTask;
    }

    // =========================================================
    // UPDATE HOOKS
    // =========================================================

    protected virtual Task BeforeUpdateAsync(
        TEntity entity,
        TUpdateDto input,
        CancellationToken cancellationToken)
    {
        return Task.CompletedTask;
    }

    protected virtual Task AfterUpdateAsync(
        TEntity entity,
        CancellationToken cancellationToken)
    {
        return Task.CompletedTask;
    }

    // =========================================================
    // DELETE HOOKS
    // =========================================================

    protected virtual Task BeforeDeleteAsync(
        TEntity entity,
        CancellationToken cancellationToken)
    {
        return Task.CompletedTask;
    }

    protected virtual Task AfterDeleteAsync(
        TEntity entity,
        CancellationToken cancellationToken)
    {
        return Task.CompletedTask;
    }

    // =========================================================
    // MAPPING
    // =========================================================

    protected virtual Task<TDto> MapToDtoAsync(
        TEntity entity,
        CancellationToken cancellationToken)
    {
        throw new NotImplementedException(
            "Override MapToDtoAsync in the derived application service.");
    }

    protected virtual Task<TEntity> MapToEntityAsync(
        TCreateDto input,
        CancellationToken cancellationToken)
    {
        throw new NotImplementedException(
            "Override MapToEntityAsync in the derived application service.");
    }

    protected virtual Task<TEntity> MapToEntityAsync(
        TUpdateDto input,
        TEntity entity,
        CancellationToken cancellationToken)
    {
        throw new NotImplementedException(
            "Override MapToEntityAsync for update in the derived application service.");
    }
}
